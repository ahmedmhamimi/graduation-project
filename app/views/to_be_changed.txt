/* ═══════ MODAL HELPERS ═══════ */
function openModal(id) {
    var modal = document.getElementById(id);
    if (modal) {
        modal.classList.remove('hidden');
        document.body.style.overflow = 'hidden';
    }
}

function closeModal(id) {
    var modal = document.getElementById(id);
    if (modal) {
        modal.classList.add('hidden');
        document.body.style.overflow = '';
    }
}

function closeModalOnBackdrop(event, id) {
    if (event.target === event.currentTarget) {
        closeModal(id);
    }
}

document.addEventListener('keydown', function(e) {
    if (e.key === 'Escape') {
        document.querySelectorAll('.fixed:not(.hidden)').forEach(function(modal) {
            if (modal.id && modal.id.startsWith('modal-')) {
                closeModal(modal.id);
            }
        });
    }
});

/* ═══════ TAB SWITCHING ═══════ */
function openTab(evt, tabName) {
    var tabContents = document.getElementsByClassName('tab-content');
    for (var i = 0; i < tabContents.length; i++) {
        tabContents[i].classList.add('hidden');
    }

    var tabLinks = document.getElementsByClassName('tab-link');
    for (var i = 0; i < tabLinks.length; i++) {
        tabLinks[i].classList.remove('border-sky-500', 'text-sky-600');
        tabLinks[i].classList.add('border-transparent', 'text-slate-500');
    }

    var tab = document.getElementById(tabName);
    if (tab) {
        tab.classList.remove('hidden');
    }
    evt.currentTarget.classList.remove('border-transparent', 'text-slate-500');
    evt.currentTarget.classList.add('border-sky-500', 'text-sky-600');

    if (history.replaceState) {
        history.replaceState(null, null, '#' + tabName);
    }
}

/* Restore tab from URL hash */
document.addEventListener('DOMContentLoaded', function() {
    var hash = window.location.hash;
    if (hash) {
        var tabName = hash.substring(1);
        var tabLinks = document.getElementsByClassName('tab-link');
        for (var i = 0; i < tabLinks.length; i++) {
            var onclick = tabLinks[i].getAttribute('onclick') || '';
            if (onclick.indexOf(tabName) !== -1 && document.getElementById(tabName)) {
                tabLinks[i].click();
                break;
            }
        }
    }
});

document.addEventListener('DOMContentLoaded', function() {
    var dataEl = document.getElementById('chart-data');
    if (!dataEl) return;

    var chartData;
    try {
        chartData = JSON.parse(dataEl.textContent);
    } catch (e) {
        console.warn('Could not parse chart data:', e);
        return;
    }

    var roleColors = ['#0284c7','#2563eb','#10b981','#f59e0b','#8b5cf6','#ef4444','#06b6d4','#ec4899'];

    /* --- Users by Role (Doughnut) --- */
    var byRole = chartData.by_role || {};
    var roleLabels = Object.keys(byRole);
    var roleValues = Object.values(byRole);
    var ctx1 = document.getElementById('chartUsersByRole');

    if (ctx1 && roleLabels.length > 0) {
        new Chart(ctx1, {
            type: 'doughnut',
            data: {
                labels: roleLabels.map(function(l) {
                    return l.charAt(0).toUpperCase() + l.slice(1);
                }),
                datasets: [{
                    data: roleValues,
                    backgroundColor: roleColors.slice(0, roleLabels.length),
                    borderWidth: 2,
                    borderColor: '#ffffff'
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: {
                        position: 'bottom',
                        labels: { font: { size: 11 }, padding: 12 }
                    }
                }
            }
        });
    }

    /* --- Students by Department (Bar) --- */
    var byDept = chartData.students_by_department || {};
    var deptLabels = Object.keys(byDept);
    var deptValues = Object.values(byDept);
    var ctx2 = document.getElementById('chartStudentsByDept');

        new Chart(ctx2, {

            options: {

                    x: { ticks: { font: { size: 10 } } }

  <script>
    function el(id) { return document.getElementById(id); }

    function setPill(name, state, text) {
      const dot = el('dot-' + name);
      const pill = el('pill-' + name);
      if (!dot || !pill) return;

      dot.className = 'dot';
      if (state === 'ok') dot.classList.add('ok');
      if (state === 'err') dot.classList.add('err');
      if (state === 'load') dot.classList.add('load');

      pill.innerHTML = '<span class="dot ' + (state || '') + '" id="dot-' + name + '"></span>' + text;
      const newDot = pill.querySelector('.dot');
      newDot.className = 'dot';
      if (state === 'ok') newDot.classList.add('ok');
      if (state === 'err') newDot.classList.add('err');
      if (state === 'load') newDot.classList.add('load');
    }

    function show(id, text, cls) {
      const e = el(id);
      e.className = 'result visible ' + cls;
      if (typeof text === 'object') text = JSON.stringify(text, null, 2);
      e.textContent = text;
    }

    function showHtml(id, html, cls) {
      const e = el(id);
      e.className = 'result visible ' + cls;
      e.innerHTML = html;
    }

    async function api(url) {
      const r = await fetch(url);
      return await r.json();
    }

    async function testHealth() {
      setPill('health', 'load', 'Health: connecting');
      show('r-health', 'Connecting...', 'load');
      try {
        const d = await api('/api/unitime/health');
        if (d.connected) {
          let t = 'STATUS: CONNECTED\n\n';
          t += `Sessions found: ${d.session_count}\n\n`;
          d.sessions.forEach(s => {
            t += `  ${s.reference}\n`;
            t += `    ${s.label}`;
            t += s.has_classes ? ' [has class data]\n\n' : ' [empty]\n\n';
          });
          setPill('health', 'ok', 'Health: connected');
          show('r-health', t, 'ok');
        } else {
          setPill('health', 'err', 'Health: disconnected');
          show('r-health', 'DISCONNECTED\n\n' + d.error, 'err');
        }
      } catch(e) {
        setPill('health', 'err', 'Health: error');
        show('r-health', 'Network error: ' + e.message, 'err');
      }
    }

    async function testSessions() {
      setPill('sessions', 'load', 'Sessions: loading');
      show('r-sessions', 'Loading...', 'load');
      try {
        const d = await api('/api/unitime/sessions');
        if (!d.success) {
          setPill('sessions', 'err', 'Sessions: error');
          show('r-sessions', d.error, 'err');
          return;
        }

        let html = `<strong>${d.count} session(s) found</strong>`;
        html += '<table class="data"><tr><th>Reference</th><th>Term</th><th>Year</th><th>Campus</th><th>Dates</th><th>Status</th></tr>';
        d.data.forEach(s => {
          const status = s.status || {};
          const cls = status.classes ? 'Classes' : '—';
          html += `<tr>
            <td><strong>${s.reference}</strong></td>
            <td>${s.term}</td>
            <td>${s.year}</td>
            <td>${s.campus}</td>
            <td>${s.beginDate} → ${s.endDate}</td>
            <td>${status.label}<br><small>${cls}</small></td>
          </tr>`;
        });
        html += '</table>';

        setPill('sessions', 'ok', `Sessions: ${d.count} loaded`);
        showHtml('r-sessions', html, 'ok');
      } catch(e) {
        setPill('sessions', 'err', 'Sessions: error');
        show('r-sessions', 'Error: ' + e.message, 'err');
      }
    }

    async function testRooms(raw) {
      const term = el('sel-rooms').value;
      setPill('rooms', 'load', 'Rooms: loading');
      show('r-rooms', 'Loading...', 'load');

      try {
        const d = await api('/api/unitime/rooms?term=' + term);
        if (!d.success) {
          setPill('rooms', 'err', 'Rooms: error');
          show('r-rooms', d.error, 'err');
          return;
        }

        if (raw) {
          setPill('rooms', 'ok', `Rooms: ${d.count} (raw)`);
          show('r-rooms', d, 'ok');
          return;
        }

        if (d.count === 0) {
          setPill('rooms', 'ok', 'Rooms: 0 found');
          show('r-rooms', `No rooms found for "${term}".`, 'ok');
          return;
        }

        let html = `<strong>${d.count} room(s) for "${term}"</strong>`;
        html += '<table class="data"><tr><th>Room</th><th>Building</th><th>Type</th><th>Capacity</th><th>Features</th></tr>';
        d.rooms.forEach(r => {
          const feats = (r.features || []).map(f => `<span class="tag">${f}</span>`).join(' ');
          html += `<tr>
            <td><strong>${r.full_name}</strong></td>
            <td>${r.building_name}</td>
            <td>${r.type}</td>
            <td>${r.capacity} (exam: ${r.exam_capacity})</td>
            <td>${feats || '—'}</td>
          </tr>`;
        });
        html += '</table>';

        setPill('rooms', 'ok', `Rooms: ${d.count} loaded`);
        showHtml('r-rooms', html, 'ok');
      } catch(e) {
        setPill('rooms', 'err', 'Rooms: error');
        show('r-rooms', 'Error: ' + e.message, 'err');
      }
    }

    async function testInstructors(raw) {
      const term = el('sel-instr').value;
      setPill('instr', 'load', 'Instructors: loading');
      show('r-instr', 'Loading...', 'load');

      try {
        const d = await api('/api/unitime/instructors?term=' + term);
        if (!d.success) {
          setPill('instr', 'err', 'Instructors: error');
          show('r-instr', d.error, 'err');
          return;
        }

        if (raw) {
          setPill('instr', 'ok', `Instructors: ${d.instructor_count} (raw)`);
          show('r-instr', d, 'ok');
          return;
        }

        if (d.instructor_count === 0) {
          setPill('instr', 'ok', 'Instructors: 0 found');
          show('r-instr', `No instructors found for "${term}".`, 'ok');
          return;
        }

        let html = `<strong>${d.instructor_count} instructor(s) across ${d.department_count} department(s)</strong>`;
        html += '<table class="data"><tr><th>Name</th><th>Position</th><th>Department</th><th>External ID</th><th>Email</th></tr>';
        d.instructors.forEach(i => {
          html += `<tr>
            <td><strong>${i.name}</strong></td>
            <td>${i.position || '—'}</td>
            <td>${i.department} (${i.department_abbr})</td>
            <td>${i.external_id || '—'}</td>
            <td>${i.email || '—'}</td>
          </tr>`;
        });
        html += '</table>';

        setPill('instr', 'ok', `Instructors: ${d.instructor_count} loaded`);
        showHtml('r-instr', html, 'ok');
      } catch(e) {
        setPill('instr', 'err', 'Instructors: error');
        show('r-instr', 'Error: ' + e.message, 'err');
      }
    }

    async function testEnrollments() {
      setPill('enroll', 'load', 'Enrollments: loading');
      show('r-enroll', 'Loading...', 'load');

      try {
        const d = await api('/api/unitime/enrollments?term=Fal2010woebegon');
        if (d.success && d.count === 0) {
          setPill('enroll', 'ok', 'Enrollments: 0 (endpoint OK)');
          show('r-enroll',
            'Endpoint works. No enrollment data loaded in UniTime.\nThis confirms the connection — data will appear once enrollments are imported.',
            'ok'
          );
        } else {
          setPill('enroll', d.success ? 'ok' : 'err', d.success ? 'Enrollments: loaded' : 'Enrollments: error');
          show('r-enroll', d, d.success ? 'ok' : 'err');
        }
      } catch(e) {
        setPill('enroll', 'err', 'Enrollments: error');
        show('r-enroll', 'Error: ' + e.message, 'err');
      }
    }
